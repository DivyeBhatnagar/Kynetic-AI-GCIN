orchestration_job.py 
import kynetic as kn

# 1. Initialize workload target
job = kn.Workload(
    model="mistralai/Mistral-7B-v0.1",
    task="fine_tune_lora",
    max_budget_usd=30.00,
    max_duration_hours=8.0
)

# 2. Compile optimal execution plan
plan = kn.planner.resolve(job)
print(f"Assigned: {plan.gpu_sku} | Est: ${plan.cost}")

# 3. Autonomous dispatch with spot fallback
cluster = kn.dispatch(plan, spot_fallback=True)
cluster.stream_telemetry()
