# Generated by Kynetic AI Workload Planner
import kynetic as ky

planner = ky.WorkloadPlanner(api_key=ky.load_env())

plan = planner.solve(
    model="meta-llama/Llama-3-70b-instruct",
    task="lora_finetune",
    sequence_length=8192,
    budget_limit_usd=50.00,
    precision="bf16",
    max_risk_index="low"
)

# Optimal Candidate match: 8x L40S via RunPod
print(f"Plan resolved: {plan.recommended_cluster} (${plan.total_cost} USD)")
job = planner.dispatch(plan, autotune_nccl=True)
job.wait_for_completion()
